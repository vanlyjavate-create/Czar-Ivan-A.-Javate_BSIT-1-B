Program Title: Daily Air Temperature Comparison

This program asks for the morning and afternoon temperatures for 7 days (from -20 to 60 degrees C) and stores them in arrays. It computes the afternoon-minus-morning difference for each day, counts the days when the afternoon was warmer, and shows the day with the greatest temperature increase.

The full file is also in this repository: TemperatureComparison.java

import java.util.Scanner;

public class TemperatureComparison {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        double[] morning = new double[7];
        double[] afternoon = new double[7];
        double[] difference = new double[7];

        // Input with validation
        for (int i = 0; i < 7; i++) {

            // Morning temperature
            boolean valid = false;
            while (valid == false) {
                System.out.print("Enter morning temperature for Day " + i + ": ");
                if (sc.hasNextDouble()) {
                    morning[i] = sc.nextDouble();
                    if (morning[i] >= -20 && morning[i] <= 60) {
                        valid = true;
                    } else {
                        System.out.println("Invalid! Temperature must be from -20 to 60.");
                    }
                } else {
                    System.out.println("Invalid! Please enter a number.");
                    sc.next();
                }
            }

            // Afternoon temperature
            valid = false;
            while (valid == false) {
                System.out.print("Enter afternoon temperature for Day " + i + ": ");
                if (sc.hasNextDouble()) {
                    afternoon[i] = sc.nextDouble();
                    if (afternoon[i] >= -20 && afternoon[i] <= 60) {
                        valid = true;
                    } else {
                        System.out.println("Invalid! Temperature must be from -20 to 60.");
                    }
                } else {
                    System.out.println("Invalid! Please enter a number.");
                    sc.next();
                }
            }
        }

        // Processing
        int warmerDays = 0;
        int maxIndex = 0;

        for (int i = 0; i < 7; i++) {
            difference[i] = afternoon[i] - morning[i];

            if (difference[i] > 0) {
                warmerDays++;
            }

            if (difference[i] > difference[maxIndex]) {
                maxIndex = i;
            }
        }

        // Output
        System.out.println();
        System.out.println("Day\tMorning\tAfternoon\tDifference");
        for (int i = 0; i < 7; i++) {
            System.out.println(i + "\t" + morning[i] + "\t" + afternoon[i] + "\t\t" + difference[i]);
        }

        System.out.println();
        System.out.println("Number of days the afternoon was warmer: " + warmerDays);
        System.out.println("Day with the greatest temperature increase: Day " + maxIndex);
        System.out.println("Increase: " + difference[maxIndex] + " degrees C");

        sc.close();
    }
}
